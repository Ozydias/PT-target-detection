### 1.1 COCO到YOLO坐标转换算法

**核心函数**：坐标转换逻辑

```python
def convert_bbox_coco_to_yolo(bbox, img_width, img_height):
    """
    将COCO格式边界框转换为YOLO格式
    
    Args:
        bbox: COCO格式边界框 [x_min, y_min, width, height] (绝对坐标)
        img_width: 图像宽度
        img_height: 图像高度
    
    Returns:
        yolo_bbox: YOLO格式边界框 [center_x, center_y, width, height] (归一化坐标)
    """
    x_min, y_min, bbox_w, bbox_h = bbox
    
    # 计算中心点坐标（归一化）
    center_x = (x_min + bbox_w / 2) / img_width
    center_y = (y_min + bbox_h / 2) / img_height
    
    # 计算归一化宽高
    norm_w = bbox_w / img_width
    norm_h = bbox_h / img_height
    
    # 边界检查
    if norm_w <= 0 or norm_h <= 0:
        return None
    
    if center_x < 0 or center_x > 1 or center_y < 0 or center_y > 1:
        return None
    
    return [center_x, center_y, norm_w, norm_h]
```

```python
def convert_coco_to_yolo(dataset_dir, train_json, val_json):
    """
    COCO格式转YOLO格式主函数
    """
    dataset_path = Path(dataset_dir)
    
    # 创建输出目录
    labels_train_dir = dataset_path / "labels" / "train"
    labels_val_dir = dataset_path / "labels" / "val"
    labels_train_dir.mkdir(parents=True, exist_ok=True)
    labels_val_dir.mkdir(parents=True, exist_ok=True)
    
    # 读取COCO格式JSON文件
    with open(dataset_path / train_json, 'r', encoding='utf-8') as f:
        train_data = json.load(f)
    
    # 建立类别ID映射
    categories = {cat['id']: cat for cat in train_data['categories']}
    cat_id_to_yolo = {cat_id: idx for idx, cat_id in enumerate(sorted(categories.keys()))}
    
    # 建立图像ID到图像信息的映射
    images = {img['id']: img for img in train_data['images']}
    
    # 建立图像ID到标注列表的映射
    img_to_anns = defaultdict(list)
    for ann in train_data['annotations']:
        img_to_anns[ann['image_id']].append(ann)
    
    # 转换训练集标注
    train_count = 0
    for img_id, anns in img_to_anns.items():
        img = images[img_id]
        h, w = img['height'], img['width']
        img_name = img['file_name']
        
        # YOLO格式标签文件路径
        label_file = labels_train_dir / (Path(img_name).stem + '.txt')
        
        # 转换标注
        yolo_labels = []
        for ann in anns:
            if ann.get('iscrowd', 0):
                continue
            
            # COCO格式: [x_min, y_min, width, height] (绝对坐标)
            bbox = np.array(ann['bbox'], dtype=np.float32)
            x_min, y_min, bbox_w, bbox_h = bbox
            
            # 转换为YOLO格式: [center_x, center_y, width, height] (归一化坐标)
            center_x = (x_min + bbox_w / 2) / w
            center_y = (y_min + bbox_h / 2) / h
            norm_w = bbox_w / w
            norm_h = bbox_h / h
            
            # 获取YOLO类别ID
            yolo_cls = cat_id_to_yolo[ann['category_id']]
            
            # 边界检查
            if norm_w <= 0 or norm_h <= 0:
                continue
            if center_x < 0 or center_x > 1 or center_y < 0 or center_y > 1:
                continue
            
            yolo_labels.append(f"{yolo_cls} {center_x:.6f} {center_y:.6f} {norm_w:.6f} {norm_h:.6f}")
        
        # 写入标签文件
        if yolo_labels:
            with open(label_file, 'w', encoding='utf-8') as f:
                f.write('\n'.join(yolo_labels))
            train_count += 1
    
    return labels_train_dir, labels_val_dir
```

```python
def check_file_matching(train_images_dir, train_labels_dir):
    """
    检查图像文件和标签文件的匹配情况
    
    Args:
        train_images_dir: 训练图像目录
        train_labels_dir: 训练标签目录
    
    Returns:
        missing_labels: 缺少标签的图像文件列表
        missing_images: 缺少图像的标签文件列表
    """
    # 获取所有图像文件名（不含扩展名）
    img_files = list(train_images_dir.glob("*.jpg")) + list(train_images_dir.glob("*.png"))
    img_names = {f.stem for f in img_files}
    
    # 获取所有标签文件名（不含扩展名）
    txt_files = list(train_labels_dir.glob("*.txt"))
    txt_names = {f.stem for f in txt_files}
    
    # 找出不匹配的文件
    missing_labels = img_names - txt_names  # 有图像但没有标签
    missing_images = txt_names - img_names  # 有标签但没有图像
    
    return missing_labels, missing_images
```

### 2.2 标签格式验证算法

```python
def validate_label_format(label_file, num_classes):
    """
    验证标签文件格式
    
    Args:
        label_file: 标签文件路径
        num_classes: 类别数量
    
    Returns:
        is_valid: 格式是否有效
        errors: 错误列表
    """
    errors = []
    
    with open(label_file, 'r') as f:
        lines = f.readlines()
    
    for line_num, line in enumerate(lines, 1):
        line = line.strip()
        if not line:
            continue
        
        parts = line.split()
        if len(parts) != 5:
            errors.append(f"第{line_num}行: 应该有5个值，实际有{len(parts)}个")
            continue
        
        try:
            class_id = int(float(parts[0]))
            center_x = float(parts[1])
            center_y = float(parts[2])
            width = float(parts[3])
            height = float(parts[4])
            
            # 检查类别ID
            if class_id < 0 or class_id >= num_classes:
                errors.append(f"第{line_num}行: 类别ID {class_id} 超出范围 [0, {num_classes-1}]")
            
            # 检查坐标范围
            if not (0 <= center_x <= 1 and 0 <= center_y <= 1):
                errors.append(f"第{line_num}行: 中心坐标超出范围 [0, 1]")
            
            if not (0 < width <= 1 and 0 < height <= 1):
                errors.append(f"第{line_num}行: 宽高超出范围 (0, 1]")
                
        except ValueError as e:
            errors.append(f"第{line_num}行: 无法解析数字 - {e}")
    
    return len(errors) == 0, errors
```

```python
def verify_p2_detection_layer(model):
    """
    验证P2检测层配置
    
    Args:
        model: YOLOv12模型
    
    Returns:
        is_valid: 配置是否有效
        details: 详细信息
    """
    detect_layer = model.model[-1]
    details = {}
    
    # 检查检测层数量
    num_layers = detect_layer.nl
    details['num_layers'] = num_layers
    details['expected_layers'] = 4  # P2, P3, P4, P5
    
    if num_layers != 4:
        return False, details
    
    # 检查stride
    if hasattr(detect_layer, 'stride'):
        strides = detect_layer.stride
        details['strides'] = strides
        
        # P2 stride应该最小（分辨率最高）
        if len(strides) >= 2 and strides[0] >= strides[1]:
            return False, details
    
    # 检查输入通道数
    if hasattr(detect_layer, 'cv2'):
        input_channels = []
        for cv in detect_layer.cv2:
            if hasattr(cv, '__len__') and len(cv) > 0:
                first_conv = cv[0]
                if hasattr(first_conv, 'conv'):
                    input_channels.append(first_conv.conv.in_channels)
        details['input_channels'] = input_channels
    
    # 测试前向传播
    dummy_input = torch.randn(1, 3, 640, 640)
    model.eval()
    with torch.no_grad():
        output = model(dummy_input)
    
    if isinstance(output, (list, tuple)) and len(output) == 4:
        details['output_shapes'] = [out.shape for out in output]
        details['forward_pass'] = True
    else:
        details['forward_pass'] = False
        return False, details
    
    return True, details
```

```python
def yolov12_inference_image(image, model_id, image_size, conf_threshold):
```

```python
def yolov12_inference_video(video_path, model_id, image_size, conf_threshold):
    """
    YOLOv12视频推理
    
    Args:
        video_path: 输入视频路径
        model_id: 模型路径或名称
        image_size: 输入图像尺寸
        conf_threshold: 置信度阈值
    
    Returns:
        output_video_path: 输出视频路径
    """
    # 加载模型
    model = YOLO(model_id)
    
    # 打开输入视频
    cap = cv2.VideoCapture(video_path)
    fps = cap.get(cv2.CAP_PROP_FPS)
    frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
    
    # 创建输出视频
    output_video_path = tempfile.mktemp(suffix=".webm")
    out = cv2.VideoWriter(
        output_video_path,
        cv2.VideoWriter_fourcc(*'vp80'),
        fps,
        (frame_width, frame_height)
    )
    
    # 逐帧处理
    while cap.isOpened():
        ret, frame = cap.read()
        if not ret:
            break
        
        # 执行预测
        results = model.predict(
            source=frame,
            imgsz=image_size,
            conf=conf_threshold
        )
        
        # 绘制检测框
        annotated_frame = results[0].plot(
            boxes=True,
            labels=True,
            conf=True,
            line_width=2
        )
        
        # 写入输出视频
        out.write(annotated_frame)
    
    # 释放资源
    cap.release()
    out.release()
    
    return output_video_path
```

```python
# 检测层初始化
class Detect(nn.Module):
    def __init__(self, nc=80, ch=()):
        super().__init__()
        self.nc = nc  # 类别数
        self.nl = len(ch)  # 检测层数量（应为4）
        self.stride = [4, 8, 16, 32]  # P2, P3, P4, P5的stride
        
        # 为每个检测层创建卷积
        self.cv2 = nn.ModuleList()
        for i in range(self.nl):
            self.cv2.append(nn.Sequential(
                Conv(c, c, 3),
                Conv(c, c, 3),
                nn.Conv2d(c, nc + 4, 1)
            ))
```

### 7.1 学习率调度算法

```python
def get_learning_rate_schedule(epoch, total_epochs, lr0, lrf, warmup_epochs):
    """
    计算当前epoch的学习率
    
    Args:
        epoch: 当前epoch
        total_epochs: 总epoch数
        lr0: 初始学习率
        lrf: 最终学习率系数
        warmup_epochs: 预热轮数
    
    Returns:
        lr: 当前学习率
    """
    # 预热阶段：线性增加
    if epoch < warmup_epochs:
        lr = lr0 * (epoch + 1) / warmup_epochs
    else:
        # 正常训练：线性衰减
        progress = (epoch - warmup_epochs) / (total_epochs - warmup_epochs)
        lr = lr0 * (1 - progress * (1 - lrf))
    
    return lr
```

## 8. 性能评估核心算法

### 8.1 mAP计算算法（简化版）

```python
def calculate_map(predictions, ground_truths, iou_threshold=0.5):
    """
    计算mAP（简化版）
    
    Args:
        predictions: 预测结果列表
        ground_truths: 真实标注列表
        iou_threshold: IoU阈值
    
    Returns:
        map_value: mAP值
    """
    # 按置信度排序
    predictions = sorted(predictions, key=lambda x: x['confidence'], reverse=True)
    
    # 计算每个类别的AP
    aps = []
    for class_id in range(num_classes):
        # 获取该类别的预测和真实标注
        class_preds = [p for p in predictions if p['class'] == class_id]
        class_gts = [g for g in ground_truths if g['class'] == class_id]
        
        # 计算TP和FP
        tp = [0] * len(class_preds)
        fp = [0] * len(class_preds)
        
        matched_gts = set()
        for i, pred in enumerate(class_preds):
            best_iou = 0
            best_gt_idx = -1
            
            for j, gt in enumerate(class_gts):
                if j in matched_gts:
                    continue
                
                iou = calculate_iou(pred['bbox'], gt['bbox'])
                if iou > best_iou:
                    best_iou = iou
                    best_gt_idx = j
            
            if best_iou >= iou_threshold:
                tp[i] = 1
                matched_gts.add(best_gt_idx)
            else:
                fp[i] = 1
        
        # 计算累积TP和FP
        tp_cumsum = np.cumsum(tp)
        fp_cumsum = np.cumsum(fp)
        
        # 计算精确率和召回率
        recalls = tp_cumsum / len(class_gts) if len(class_gts) > 0 else np.zeros_like(tp_cumsum)
        precisions = tp_cumsum / (tp_cumsum + fp_cumsum)
        
        # 计算AP（使用11点插值法）
        ap = 0
        for t in np.arange(0, 1.1, 0.1):
            if np.sum(recalls >= t) == 0:
                p = 0
            else:
                p = np.max(precisions[recalls >= t])
            ap += p / 11
        
        aps.append(ap)
    
    # 计算mAP
    map_value = np.mean(aps)
    return map_value
```
